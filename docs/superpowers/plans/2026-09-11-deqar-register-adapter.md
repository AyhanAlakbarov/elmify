# DEQAR and WHED Register Adapters Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn DEQAR's four daily files into `QUALITY_ASSURED` rows and WHED's per-institution pages into `NATIONALLY_RECOGNISED` rows in `RegisterListing`, on machinery every later register reuses.

**Architecture:** Each register is a config: a `collect` step that fetches its pages and a pure `read` that turns bytes into verdicts, with no database or network in the read. One shared writer joins verdicts to the catalog through `InstitutionIdentifier` and replaces the register's rows in one transaction. One shared orchestrator runs collect, origin check, read, archive and record, in that order, and one management command wraps it.

**Tech Stack:** Python 3.14, Django 5.2, PostgreSQL with RLS, `csv` and `lxml` (already a dependency), pytest with pytest-django, `uv`.

**Spec:** `docs/superpowers/specs/2026-09-11-deqar-register-adapter-design.md`, with the WHED site measurements and terms in `docs/superpowers/research/2026-09-11-whed-recon.md`. Read both first; they carry the reasons this plan does not repeat.

```python
class OffSiteRedirect(ValueError):
    """The fetch ended somewhere other than the source the catalog vouched for.

    Why origin rather than the whole URL, and why this must not be tightened:
    CLAUDE.md, "Decision B vouches for the SOURCE ROW, not the BYTES". It lives
    here rather than in refresh.py because the register path uses it too, and
    refresh.py imports the model client that path is pinned never to reach.
    """


def _origin(url: str) -> tuple[str, str]:
    """Scheme and host, lowercased -- the part a redirect must not change."""
    parsed = urlsplit(url)
    return parsed.scheme.lower(), (parsed.hostname or "").lower()


def require_same_origin(requested: str, fetched: FetchResult) -> None:
    """Raise OffSiteRedirect unless the fetch ended on the requested origin.

    Call it before persist_snapshot: off-site bytes must not be archived under
    the source's provenance either.
    """
    if _origin(fetched.final_url) != _origin(requested):
        raise OffSiteRedirect(f"{requested} redirected off-site to {fetched.final_url}")
```

```python
def _csv(name: str, rows: list[dict[str, str]]) -> bytes:
    out = io.StringIO()
    writer = csv.DictWriter(out, fieldnames=_HEADERS[name].split(","), restval="")
    writer.writeheader()
    writer.writerows(rows)
    # DEQAR ships a UTF-8 byte-order mark; the read must survive it.
    return ("\ufeff" + out.getvalue()).encode("utf-8")
```

```python
def _rows(pages: Mapping[str, bytes], name: str) -> csv.DictReader[str]:
    """One DEQAR file as rows, refused if a column the read uses is missing."""
    if name not in pages:
        raise UnreadableRegister(f"{name}: file missing")
    # A stream, not one decoded str: a single character outside Latin-1 makes
    # CPython store the whole 103 MB reports file at two or four bytes per
    # character. newline="" because quoted fields carry line breaks.
    stream = io.TextIOWrapper(io.BytesIO(pages[name]), encoding="utf-8-sig", newline="")
    reader = csv.DictReader(stream)
    missing = [column for column in _DEQAR_COLUMNS[name] if column not in (reader.fieldnames or ())]
    if missing:
        raise UnreadableRegister(f"{name}: missing column(s) {missing}")
    return reader


def _report_end(row: Mapping[str, str]) -> tuple[date, bool] | None:
    """When a report stops counting, and whether that date is DEQAR's six-year rule.

    docs.deqar.eu/stable/report_data/: "In case the end date is left open, the
    report will be treated as valid for six years from the start of its
    validity." None means the dates do not parse, and such a report does not
    count -- the cautious direction, since QUALIFIED is never shown as a fail.
    """
    try:
        if row["report_valid_to"]:
            return date.fromisoformat(row["report_valid_to"]), False
        start = date.fromisoformat(row["report_valid_from"])
    except ValueError:
        return None
    try:
        return start.replace(year=start.year + 6), True
    except ValueError:  # 29 February, six years on, is not a date.
        return start.replace(year=start.year + 6, day=28), True


def _describe(row: Mapping[str, str], end: tuple[date, bool] | None, *, in_force: bool) -> str:
    """One report in DEQAR's own field values; the joining words are the only ones we add."""
    head = f"{row['report_agency']} {row['report_type']} report, {row['report_decision']}"
    if end is None:
        tail = "validity unreadable"
    elif in_force:
        tail = f"valid to {end[0].isoformat()}" + (" (six years from start)" if end[1] else "")
    else:
        tail = f"expired {end[0].isoformat()}"
    return f"{head}, {tail}"[:_RAW_VALUE_MAX]

    QUALIFIED: registered-agency reports exist and none counts. No row:
    only lapsed agencies have reported, or nobody has. Never EXCLUDED --

    docs/superpowers/specs/2026-09-11-deqar-register-adapter-design.md §3.1.

        row["acronym_primary"]
        for row in _rows(pages, "agencies")
        if row["is_registered"] == "Registered"
    }

        raise UnreadableRegister("agencies: no agency reads 'Registered'")

    iso_by_name = {
        row["name_english"]: row["iso_3166_alpha2"][:2].upper()
        for row in _rows(pages, "countries")
        if row["iso_3166_alpha2"]
    }

    qualified: dict[str, tuple[date, str]] = {}
    report_rows = 0
    for row in _rows(pages, "reports"):
        report_rows += 1
        if row["report_agency"] not in registered:
            continue
        end = _report_end(row)
        in_force = end is not None and end[0] >= today
        hei = row["hei_deqar_id"]
        described = _describe(row, end, in_force=in_force)
        if end is not None and in_force and row["report_decision"] != "negative":
            rank = (row["report_type"] in _INSTITUTIONAL, end[0])
            if hei not in clear or rank > clear[hei][0]:
                clear[hei] = (rank, described)
        else:
            latest = end[0] if end is not None else date.min
            if hei not in qualified or latest > qualified[hei][0]:
                qualified[hei] = (latest, described)
    if not report_rows:
        raise UnreadableRegister("reports: no rows")

    open_ids: set[str] = set()
    counted: set[tuple[str, str]] = set()
    listed: Counter[str] = Counter()
    cleared: Counter[str] = Counter()
    unmapped: set[str] = set()
    crosswalk: dict[str, tuple[tuple[str, str], ...]] = {}
    institution_rows = 0
    for row in _rows(pages, "institutions"):
        institution_rows += 1
        if row["closure_date"]:
            continue
        deqar_id, country = row["deqar_id"], row["country"]
        open_ids.add(deqar_id)
        whed_ids = _DEQAR_WHED_ID.findall(row["identifiers_all"])
        if whed_ids and deqar_id not in crosswalk:
            crosswalk[deqar_id] = tuple(("WHED", value) for value in whed_ids)
        if not country:
            continue
        code = iso_by_name.get(country) or DEQAR_COUNTRY_NAMES.get(country, "")
        if code not in ISO_3166_ALPHA2_CODES:
            unmapped.add(country)
            continue
        if (deqar_id, code) in counted:
            continue
        counted.add((deqar_id, code))
        listed[code] += 1
        cleared[code] += int(deqar_id in clear)
    if not institution_rows:
        raise UnreadableRegister("institutions: no rows")

    listings: dict[str, ProposedListing] = {}
    for deqar_id in open_ids:
        if deqar_id in clear:
            listings[deqar_id] = ProposedListing(
                RegisterListing.Verdict.CLEAR.value, clear[deqar_id][1], page="reports"
            )
        elif deqar_id in qualified:
            listings[deqar_id] = ProposedListing(
                RegisterListing.Verdict.QUALIFIED.value, qualified[deqar_id][1], page="reports"
            )
    return RegisterRead(
        scheme=InstitutionIdentifier.Scheme.DEQARINST.value,
        axis=RegisterListing.Axis.QUALITY_ASSURED.value,
        as_of=today,
        listings=listings,
        coverage=tuple(
            ProposedCoverage(code, listed[code], cleared[code]) for code in sorted(listed)
        ),
        coverage_page="reports",
        crosswalk=crosswalk,
        unmapped_countries=tuple(sorted(unmapped)),
    )
```

```python
#: Most cautious wins -- the same order a read applies across registers.
_CAUTION = {"CLEAR": 0, "QUALIFIED": 1, "EXCLUDED": 2}


@dataclass(frozen=True)
class RecordSummary:
    written: int
    #: Listings whose identifier is bound to no catalog institution.
    dropped: int
    coverage: int
    #: Bound institutions whose verdict differs from the previous read,
    #: including one appearing or disappearing. Stands in for the design's
    #: change alert until that lands.
    changed: int
    #: Crosswalk codes newly bound, and codes found bound to a different
    #: institution -- a duplicate catalog row or a wrong binding, for a human.
    crosswalk_written: int
    crosswalk_conflicts: tuple[str, ...]


@transaction.atomic
def record_register_read(
    *,
    publisher: str,
    read: RegisterRead,
    snapshots: Mapping[str, OfficialSourceSnapshot],
) -> RecordSummary:
    """Replace this publisher's rows with one read.

    Replace rather than merge: the table is a projection of the newest read,
    and the archived snapshots are the record of every read before it. The
    scope is the publisher, not one source, because WHED archives one source
    per institution page.

    Rows join on InstitutionIdentifier and nothing else. A value bound to no
    institution is dropped, never matched on a name.

    Writes need the catalog_ingestion policy. Under the runtime role the
    DELETE matches nothing and the INSERT raises, which rolls the DELETE back
    with it: a wrong-role run fails loudly and changes nothing.
    """
    bound: dict[str, UUID] = dict(
        InstitutionIdentifier.objects.filter(
            scheme=read.scheme, value__in=list(read.listings)
        ).values_list("value", "institution_id")
    )
    new: dict[UUID, ProposedListing] = {}
    for value, listing in read.listings.items():
        institution_id = bound.get(value)
        if institution_id is None:
            continue
        held = new.get(institution_id)
        # One institution may hold two identifiers after a merger.
        if held is None or _CAUTION[listing.verdict] > _CAUTION[held.verdict]:
            new[institution_id] = listing

    mine = RegisterListing.objects.filter(snapshot__source__publisher=publisher)
    previous: dict[UUID, str] = dict(mine.values_list("institution_id", "verdict"))
    mine.delete()
    RegisterCoverage.objects.filter(snapshot__source__publisher=publisher).delete()

    RegisterListing.objects.bulk_create(
        RegisterListing(
            institution_id=institution_id,
            snapshot=snapshots[listing.page],
            axis=read.axis,
            verdict=listing.verdict,
            raw_value=listing.raw_value,
            as_of=read.as_of,
        )
        for institution_id, listing in new.items()
    )
    if read.coverage:
        RegisterCoverage.objects.bulk_create(
            RegisterCoverage(
                snapshot=snapshots[read.coverage_page],
                axis=read.axis,
                country_code=row.country_code,
                listed=row.listed,
                cleared=row.cleared,
            )
            for row in read.coverage
        )

    crosswalk_written = 0
    conflicts: list[str] = []
    crosswalk_bound = InstitutionIdentifier.objects.filter(
        scheme=read.scheme, value__in=list(read.crosswalk)
    ).values_list("value", "institution_id")
    for value, institution_id in crosswalk_bound:
        for scheme, code in read.crosswalk[value]:
            row, created = InstitutionIdentifier.objects.get_or_create(
                scheme=scheme,
                value=code,
                defaults={
                    "institution_id": institution_id,
                    "source": InstitutionIdentifier.Source.CROSSWALK.value,
                },
            )
            if created:
                crosswalk_written += 1
            elif row.institution_id != institution_id:
                conflicts.append(f"{scheme}:{code}")

    changed = sum(
        1
        for institution_id in previous.keys() | new.keys()
        if previous.get(institution_id)
        != (new[institution_id].verdict if institution_id in new else None)
    )
    return RecordSummary(
        written=len(new),
        dropped=sum(1 for value in read.listings if value not in bound),
        coverage=len(read.coverage),
        changed=changed,
        crosswalk_written=crosswalk_written,
        crosswalk_conflicts=tuple(sorted(conflicts)),
    )
```

```python
@dataclass(frozen=True)
class ArchivedPage:
    snapshot: OfficialSourceSnapshot
    #: False when the bytes matched a snapshot already archived.
    created: bool


@dataclass(frozen=True)
class RefreshReport:
    pages: Mapping[str, ArchivedPage]
    read: RegisterRead
    summary: RecordSummary


def _archive(register: Register, page: CollectedPage, store: ObjectStore) -> ArchivedPage:
    source, _created = OfficialSource.objects.get_or_create(
        canonical_url=page.url,
        defaults={"publisher": register.publisher, "source_type": OfficialSource.Type.REGULATOR},
    )
    # (source, sha256) is unique, so archiving identical bytes twice raises.
    # The same lookup pipeline.py makes before persist_snapshot.
    digest = hashlib.sha256(page.fetched.body).hexdigest()
    existing = OfficialSourceSnapshot.objects.filter(source=source, sha256=digest).first()
    if existing is not None:
        return ArchivedPage(existing, created=False)
    # A register is not cycle-bound, so admissions_cycle is empty.
    result = persist_snapshot(
        source=source, fetch=page.fetched, store=store, admissions_cycle=""
    )
    return ArchivedPage(result.snapshot, created=True)


def refresh_register(
    register: Register,
    *,
    store: ObjectStore,
    today: date,
    fetch: Callable[..., FetchResult] = fetch_official_source,
) -> RefreshReport:
    """Collect, check, read, archive and record one register, in that order.

    The order is the guarantee. Every page is fetched and origin-checked
    before anything is archived, and read before anything is replaced: an
    off-site redirect archives nothing, and an unreadable page leaves the
    previous rows standing. The read runs even when every page is
    byte-identical to the last run, because a verdict depends on today's date
    as well as the bytes -- a report can expire overnight in a file that did
    not change.

    `fetch` is a seam for tests and for callers in this codebase. Never bind
    it from settings, a task argument or a dotted path: refresh.py states the
    same rule for its own `retrieve=`.
    """

    def checked(url: str) -> FetchResult:
        result = fetch(
            url,
            require_https=True,
            content_types=register.content_types,
            max_bytes=register.max_bytes,
        )
        require_same_origin(url, result)
        return result

    collected = register.collect(checked)
    read = register.read({name: page.fetched.body for name, page in collected.items()}, today)
    archived = {name: _archive(register, page, store) for name, page in collected.items()}
    summary = record_register_read(
        publisher=register.publisher,
        read=read,
        snapshots={name: page.snapshot for name, page in archived.items()},
    )
    return RefreshReport(pages=archived, read=read, summary=summary)
```

```python
"""Fetch one government register and replace what it says.

    DATABASE_ROLE=migration uv run python manage.py refresh_register deqar
    DATABASE_ROLE=migration uv run python manage.py refresh_register whed

The register tables accept writes only under the catalog_ingestion policy, and
DATABASE_ROLE offers runtime or migration, so a manual run uses migration, as
load_institution_identifiers does. Under the runtime role the first INSERT
raises and the DELETE before it rolls back with it: a wrong-role run fails
loudly and changes nothing. A scheduled job should not hold migration
credentials; adding a catalog_ingestion DATABASE_ROLE is part of scheduling
this, not of this command.

Design: docs/superpowers/specs/2026-09-11-deqar-register-adapter-design.md.

from __future__ import annotations

from collections import Counter

from django.core.management.base import BaseCommand, CommandError, CommandParser
from django.utils import timezone

from apps.research.egress import OffSiteRedirect, SourceBlocked, UnsafeTarget
from apps.research.registers import REGISTERS, UnreadableRegister, refresh_register

class Command(BaseCommand):
    help = "Fetch a government register and replace its listings and coverage"

    def add_arguments(self, parser: CommandParser) -> None:
        parser.add_argument("slug", choices=sorted(REGISTERS))

    def handle(self, *args: object, **options: object) -> None:
        register = REGISTERS[str(options["slug"])]
        try:
            report = refresh_register(
                register, store=S3ObjectStore(), today=timezone.localdate()
            )
        except (UnreadableRegister, OffSiteRedirect, SourceBlocked, UnsafeTarget) as error:
            raise CommandError(f"{register.slug}: {error}") from error

        slug = register.slug
        for name, archived in report.pages.items():
            state = "created" if archived.created else "unchanged, reused"
            self.stdout.write(
                f"{slug}: {name} {archived.snapshot.byte_size} bytes, snapshot {state}"
            )
        verdicts = Counter(listing.verdict for listing in report.read.listings.values())
        self.stdout.write(
            f"{slug}: "
            + ", ".join(f"{verdict} {count}" for verdict, count in sorted(verdicts.items()))
        )
        summary = report.summary
        self.stdout.write(
            f"{slug}: written {summary.written}, unbound {summary.dropped}, "
            f"coverage rows {summary.coverage}, changed since last read {summary.changed}"
        )
        self.stdout.write(
            f"{slug}: crosswalk bound {summary.crosswalk_written}, "
            f"crosswalk conflicts {len(summary.crosswalk_conflicts)}"
            + (": " + ", ".join(summary.crosswalk_conflicts) if summary.crosswalk_conflicts else "")
        )
        if report.read.unmapped_countries:
            self.stdout.write(
                f"{slug}: unmapped countries, left out of coverage: "
                + ", ".join(report.read.unmapped_countries)
            )
```

```python
#: The bare host only. www.whed.net answers every path with a Cloudflare
#: challenge; whed.net serves the same pages plainly. Measured 2026-09-11 and
#: 2026-09-12, docs/superpowers/research/2026-09-11-whed-recon.md.
_WHED_SEARCH = "https://whed.net/results_institutions.php"
_WHED_PACE_SECONDS = 1.0


def _whed_search_url(iau_id: str) -> str:
    """WHED's own search for one Global WHED ID, as the site's form submits it.

    The search answers "is this ID in the list now", which is WHED's listing
    rule directly. An institution's own page answers only "does a page exist",
    and whether a delisted institution's page is kept has never been observed.
    """
    query = urlencode(
        (
            ("Chp2", iau_id),
            ("Chp0", ""),
            ("Chp1", ""),
            ("Ope0", "and"),
            ("Chp4", ""),
            ("Chp5", ""),
            ("nbr_ref_pge", "100"),
            ("debut", "0"),
            ("afftri", "yes"),
            ("stat", "Country"),
        )
    )
    return f"{_WHED_SEARCH}?{query}"


def _whed_result(body: bytes, iau_id: str) -> str | None:
    """The listing's raw_value if the search found the ID, None if it did not.

    Keys on the whole result line, which echoes the searched ID, and never on
    the ID's presence: the no-match page prints the ID too, in that line and in
    the search box. Anything other than the two measured answers refuses.
    """
    try:
        document = lxml.html.document_fromstring(body)
    except (ParserError, ValueError):
        raise UnreadableRegister(f"{iau_id}: search page does not parse") from None
    lines = [" ".join(p.text_content().split()) for p in document.xpath("//p[@class='infos']")]
    criterion = f"(Global WHED ID={iau_id})"
    if lines == [f"No matching records for {criterion}"]:
        return None
    if lines != [f"1 results found for {criterion}"]:
        raise UnreadableRegister(f"{iau_id}: unexpected search answer {lines[:1]!r}")
    rows = [" ".join(s.text_content().split()) for s in document.xpath("//span[@class='gui']")]
    if rows != [iau_id]:
        raise UnreadableRegister(f"{iau_id}: search row does not carry it {rows[:3]!r}")
    names = document.xpath("//ul[@id='results']//h3/a")
    countries = document.xpath("//ul[@id='results']//div[@align='right']")
    name = " ".join(names[0].text_content().split()) if names else ""
    country = " ".join(countries[0].text_content().split()) if countries else ""
    if not name:
        return "Listed"
    return (f"Listed as {name}" + (f", {country}" if country else ""))[:_RAW_VALUE_MAX]


def read_whed(pages: Mapping[str, bytes], today: date) -> RegisterRead:
    """One WHED search per bound ID -> NATIONALLY_RECOGNISED, CLEAR when listed.

    WHED lists only institutions the national competent body recognises and
    removes them when that lapses, so the search's own answer is the verdict.
    No match makes no row. Any other answer refuses the whole read, and so does
    finding none of a non-empty set of bound IDs -- a changed search must not
    read as every institution delisted at once. Never QUALIFIED, never
    EXCLUDED. Spec §3.3.
    """
    listings: dict[str, ProposedListing] = {}
    for iau_id, body in pages.items():
        raw_value = _whed_result(body, iau_id)
        if raw_value is not None:
            listings[iau_id] = ProposedListing(
                RegisterListing.Verdict.CLEAR.value, raw_value, page=iau_id
            )
    if pages and not listings:
        raise UnreadableRegister("whed: none of the bound institutions was found")
    return RegisterRead(
        scheme=InstitutionIdentifier.Scheme.WHED.value,
        axis=RegisterListing.Axis.NATIONALLY_RECOGNISED.value,
        as_of=today,
        listings=listings,
    )


def _collect_whed(
    fetch: Fetch, *, sleep: Callable[[float], None] = time.sleep
) -> dict[str, CollectedPage]:
    """One search per WHED ID the catalog has bound, paced one a second.

    The country listing is not walked: the adapter answers for institutions
    the catalog knows, and proposing new bindings is the identifier design's
    propose command, not this one.
    """
    values = list(
        InstitutionIdentifier.objects.filter(scheme=InstitutionIdentifier.Scheme.WHED.value)
        .order_by("value")
        .values_list("value", flat=True)
    )
    pages: dict[str, CollectedPage] = {}
    for position, value in enumerate(values):
        if position:
            sleep(_WHED_PACE_SECONDS)
        url = _whed_search_url(value)
        pages[value] = CollectedPage(url, fetch(url))
    return pages
```
