```python
"""What one card needs from its referees (lanes spec §11.1).

Pure: the card's copied `referee_requirements` records in, one RefereeNeed out.
A record is `{"field", "value", "date", "conditional", "condition", "rule_text",
"source_url"}` (apps/documents/rule_sinks.py::to_referee). A field with no record
was not stated, and stays None here: the page says "Not stated — check the page",
never a default.
"""

from __future__ import annotations

from collections.abc import Sequence
from dataclasses import dataclass

#: rule_sinks writes these five fields (grammar.REFEREE_FIELDS).
FIELDS = ("count", "kind", "mode", "institutional_email_required", "due")
#: count -> letters. THREE_OR_MORE is read as its floor: three.
_LETTERS = {"ONE": 1, "TWO": 2, "THREE_OR_MORE": 3, "NONE_REQUIRED": 0}
#: The only condition kind a card can answer: "scholarship applicants also need..."
#: (grammar.CONDITION_KINDS). nationality, degree_level, work_experience and other
#: need the student's facts, which this function does not hold; such a record never
#: applies here and is shown beside the need instead.
_TRACK = "track"
NOT_STATED = "Not stated — check the page"


@dataclass(frozen=True)
class Stated:
    value: str
    rule_text: str


@dataclass(frozen=True)
class RefereeNeed:
    count: Stated | None = None
    kind: Stated | None = None
    mode: Stated | None = None
    institutional_email_required: Stated | None = None
    due: Stated | None = None
    #: Conditional sentences that did not apply to this card, shown as they are.
    other_conditions: tuple[str, ...] = ()

    @property
    def letters(self) -> int | None:
        """How many letters, or None when the page does not say."""
        return _LETTERS.get(self.count.value) if self.count is not None else None

    @property
    def institutional_email(self) -> bool:
        return (
            self.institutional_email_required is not None
            and self.institutional_email_required.value == "YES"
        )


def _applies(record: dict[str, object], *, route_kind: str, award: bool) -> bool:
    if not record.get("conditional"):
        return True
    if record.get("condition") != _TRACK:
        return False
    # A track condition is a funded route: the card is a scholarship's own, or its
    # route is not the self-funded one.
    return award or (bool(route_kind) and route_kind != "SELF_FUNDED")


def needs_for(
    records: Sequence[dict[str, object]], *, route_kind: str = "", award: bool = False
) -> RefereeNeed:
    """One need per card. For each field the first applicable conditional record wins
    over an unconditional one (it is the rule for this route); otherwise the first."""
    chosen: dict[str, Stated] = {}
    specific: set[str] = set()
    others: list[str] = []
    for record in records:
        if not isinstance(record, dict):
            continue
        field, value = str(record.get("field", "")), str(record.get("value", ""))
        if field not in FIELDS or not value or value == "NOT_STATED":
            continue
        text = str(record.get("rule_text", ""))
        if not _applies(record, route_kind=route_kind, award=award):
            if text and text not in others:
                others.append(text)
            continue
        conditional = bool(record.get("conditional"))
        if field in specific or (field in chosen and not conditional):
            continue
        chosen[field] = Stated(value, text)
        if conditional:
            specific.add(field)
    return RefereeNeed(**chosen, other_conditions=tuple(others))
```

```python
USING (tenant_id = NULLIF(current_setting('app.tenant_id', true), '')::uuid)

ALTER TABLE documents_refereerequest ENABLE ROW LEVEL SECURITY;

USING (tenant_id = NULLIF(current_setting('app.tenant_id', true), '')::uuid)
```

```python
_TOO_MANY = "Too many changes at once. Try again in a while."


class RefereeForm(forms.ModelForm):  # type: ignore[type-arg]
    class Meta:
        model = Referee
        fields = (
            "title", "name", "institution", "email", "institutional_email", "kind",
            "relationship", "agreed_requests", "gives_generic_letter",
        )

def cv_referees(student: Membership, referees: list[Referee]) -> list[VerifiedFact]:
    """References read from the student's CV that are not referees yet: offered as a
    starting list, never added on their own. A reference the CV says not to contact
    is never offered."""
    known = {r.email.casefold() for r in referees if r.email} | {
        r.name.casefold() for r in referees
    }
    offered = []
    for fact in unsuperseded_facts(student).filter(category=VerifiedFact.Category.REFERENCE):
        payload = fact.payload if isinstance(fact.payload, dict) else {}
        name, email = str(payload.get("name") or ""), str(payload.get("email") or "")
        if not name or payload.get("do_not_contact"):
            continue
        if name.casefold() in known or (email and email.casefold() in known):
            continue
        offered.append(fact)
    return offered


def referee_context(student: Membership, *, counselor_view: bool) -> dict[str, Any]:
    referees = list(
        Referee.objects.filter(student=student)
        .prefetch_related("requests__application__candidate")
        .order_by("created_at")
    )
    return {
        "referees": referees,
        # One edit form per professor, the student's only; auto_id keeps the ids apart.
        "rows": [
            (r, None if counselor_view else RefereeForm(instance=r, auto_id=f"id_{r.id}_%s"))
            for r in referees
        ],
        "counselor_view": counselor_view,
        "from_cv": [] if counselor_view else cv_referees(student, referees),
        "statuses": Referee.Status.choices,
    }


def _page(
    request: HttpRequest, student: Membership, form: RefereeForm, *, status: int = 200
) -> HttpResponse:

    return render(request, "student/referees.html", context, status=status)


def _limited(request: HttpRequest, student: Membership) -> bool:
    if rate_limited(
        "referee-edit", student.id, limit=_LIMIT, window_seconds=_WINDOW_SECONDS
    ):
        messages.error(request, _TOO_MANY)
        return True
    return False

    tab waits here instead of racing (apps/research/wallet.py::open_wallet). Call
    inside transaction.atomic."""
    Membership.objects.select_for_update().get(id=student.id)

@login_required
@require_POST
def referee_add(request: HttpRequest) -> HttpResponse:
    student = current_membership(request, Membership.Role.STUDENT)
    if _limited(request, student):
        return redirect("student-referees")
    form = RefereeForm(request.POST)
    if not form.is_valid():
        return _page(request, student, form, status=400)
    with transaction.atomic():
        _lock(student)
        name, email = form.cleaned_data["name"], form.cleaned_data["email"]
        if Referee.objects.filter(
            student=student, name__iexact=name, email__iexact=email
        ).exists():
            # A double click, or the same professor typed twice.
            messages.info(request, f"{name} is already on your list.")
            return redirect("student-referees")
        referee = form.save(commit=False)
        referee.student = student
        referee.save()
        record_event(actor=student, event="referee.added", object_id=referee.id)
    return redirect("student-referees")


@login_required
@require_POST
def referee_from_cv(request: HttpRequest, fact_id: UUID) -> HttpResponse:
    student = current_membership(request, Membership.Role.STUDENT)
    fact = get_object_or_404(
        VerifiedFact, id=fact_id, student=student, category=VerifiedFact.Category.REFERENCE,
        superseded_by__isnull=True,
    )
    payload = fact.payload if isinstance(fact.payload, dict) else {}
    if not payload.get("name") or payload.get("do_not_contact"):
        raise Http404  # never offered (cv_referees), so never added
    if _limited(request, student):
        return redirect("student-referees")
    with transaction.atomic():
        _lock(student)
        # Re-checked under the lock: a second click finds them added and changes nothing.

            return redirect("student-referees")
        referee = Referee.objects.create(
            student=student,
            name=str(payload["name"])[:200],
            title=str(payload.get("position") or "")[:60],
            institution=str(payload.get("institution") or "")[:300],
            email=str(payload.get("email") or "")[:320],
        )
        record_event(actor=student, event="referee.added", object_id=referee.id)

        return redirect("student-referees")

        return redirect("student-referees")
    referee.status = status
    referee.save(update_fields=["status", "updated_at"])
    record_event(actor=student, event="referee.status", object_id=referee.id)
    return redirect("student-referees")
```

  - `referee_views.repropose(student: Membership) -> Proposal`;
  - URL names `student-referee-plan`, `student-referee-propose`, `student-referee-request-add`, `student-referee-request-remove`;
  - audit events `referee.plan_proposed`, `referee.request_added`, `referee.request_removed`.

Decisions, each pinned by a test below:
- **"Strongest"**, from stored data only: an academic referee before a professional one, then the one who agreed to more requests, then by name, so the result is stable. **"Funding-deciding"** is a scholarship's own card (`candidate.opportunity_id` set): no other stored fact says an application decides funding.
- **What counts against `agreed_requests`:** `PORTAL_INVITE` and `REFEREE_SENDS`, as the spec says, **and an unstated mode**. An unstated mode may be a portal invitation, and a professor is never asked for more than they agreed to. Uploads never count.
- **Distinct referees per application.** `letters` is met by different people, never one professor twice.
- **A count the page does not state proposes nothing.** It is listed in `Proposal.unstated`, and the page shows "Not stated — check the page" (no default).
- **"Confirm" is not a state.** The student edits the proposal (remove a proposed row, add a professor to an application), and Task 5's "I've asked" (`ASKED`) is the confirmation. Spec §3.2's state list has no confirmed state, so none is added.
- **Swap referee** is the status form on the referees page. Marking a professor `DECLINED` or `SILENT` deletes their `PROPOSED` rows, sets their `ASKED`/`AGREED` rows to `DECLINED`, keeps `SUBMITTED`, and proposes again. A `DECLINED` (application, professor) pair is never proposed again **by the platform**. The student can put it back: a silent professor who answers is set back to "Writing for you", and "Add a referee" on that application turns the `DECLINED` row back into `PROPOSED` (the unique pair would otherwise refuse it forever).
- **Only applications still waiting for a decision are planned.** `_open_applications` keeps cards with no `outcome` (a withdrawn, admitted, waitlisted or rejected card needs no new letter). Their `ASKED`/`AGREED`/`SUBMITTED` requests still count against the professor's agreement, so a professor is never asked for more than they agreed to.
- **Two tabs or a double click on "Propose" or "Add".** `repropose` and `referee_request_add` take `_lock` first. Without it, two concurrent proposals both delete the old `PROPOSED` rows and both insert the same pairs, and the second dies on the unique pair with a 500.
- **The "too few" sentence** keeps the spec's shape. The generic clause names the professor ("can use Prof. Aliyev's generic one"), because no gender is stored to write "his". It counts the programmes whose letter mode is `UPLOAD_GENERIC`. See Notes from writing.

```python
"""Who writes for which application (lanes spec §11.2). Pure: needs and referees in,
a proposal out; the page writes it as PROPOSED RefereeRequest rows.

Rules, in the spec's order:
- each application gets its `letters`, from distinct referees of its `kind`;
- a referee's PORTAL_INVITE and REFEREE_SENDS requests stay within `agreed_requests`
  (an unstated mode counts too: it may be a portal invitation, and a professor is
  never asked for more than they agreed to);
- a generic letter serves every UPLOAD_GENERIC application at no cost;
- the strongest referee goes first to funding-deciding applications, then to the
  earliest deadlines;
- an institutional address is preferred where one is required, and flagged when
  the referee given has none.

"Strongest", from what is stored: an academic referee before a professional one,
then the one who agreed to more requests, then by name so the result is stable.
"""

from __future__ import annotations

from collections.abc import Sequence
from dataclasses import dataclass, field
from datetime import datetime
from uuid import UUID

#: The modes that cost one of a referee's agreed requests.
_UPLOADS = frozenset({"UPLOAD_GENERIC", "UPLOAD_ADDRESSED"})
_ORDINALS = ("first", "second", "third", "fourth", "fifth")


def costs_a_request(mode: str) -> bool:
    return mode not in _UPLOADS


@dataclass(frozen=True)
class ApplicationNeed:
    application_id: UUID
    title: str
    #: None when the page does not say how many: nothing is proposed for it.
    letters: int | None
    #: ACADEMIC, PROFESSIONAL, EITHER, or "" (not stated: anyone).
    kind: str = ""
    #: The referee mode copied from the requirement, or "" (not stated).
    mode: str = ""
    institutional_email: bool = False
    deadline: datetime | None = None
    funding_deciding: bool = False


@dataclass(frozen=True)
class RefereeFacts:
    referee_id: UUID
    #: How the student names them: "Prof. Aliyev".
    name: str
    kind: str
    agreed_requests: int
    gives_generic_letter: bool = False
    institutional_email: bool = False


@dataclass(frozen=True)
class Assignment:
    application_id: UUID
    referee_id: UUID
    mode: str
    #: Institutional address required, and this referee's is not one.
    email_flag: bool = False


@dataclass(frozen=True)
class Proposal:
    assignments: tuple[Assignment, ...]
    #: The "too few" sentences, in the spec's shape.
    short: tuple[str, ...] = ()
    #: Applications whose page does not say how many letters.
    unstated: tuple[UUID, ...] = ()


@dataclass
class _Load:
    used: int = 0
    on: set[UUID] = field(default_factory=set)


def _strength(referee: RefereeFacts) -> tuple[bool, int, str]:
    return (referee.kind != "ACADEMIC", -referee.agreed_requests, referee.name)


def _fits(referee: RefereeFacts, need: ApplicationNeed) -> bool:
    return need.kind in ("", "EITHER") or referee.kind == need.kind


def _order(need: ApplicationNeed) -> tuple[bool, bool, datetime]:
    return (not need.funding_deciding, need.deadline is None, need.deadline or datetime.max)


def propose(
    needs: Sequence[ApplicationNeed],
    referees: Sequence[RefereeFacts],
    *,
    kept: Sequence[Assignment] = (),
    declined: Sequence[tuple[UUID, UUID]] = (),
) -> Proposal:
    """`kept` are requests already ASKED, AGREED or SUBMITTED: they fill their slots
    and use their referee's requests before anything new is proposed. `declined` are
    (application id, referee id) pairs a professor already said no to: never proposed
    again."""
    blocked = set(declined)
    load = {referee.referee_id: _Load() for referee in referees}
    modes = {need.application_id: need.mode for need in needs}
    for done in kept:
        if done.referee_id in load:
            load[done.referee_id].on.add(done.application_id)
            if costs_a_request(modes.get(done.application_id, done.mode)):
                load[done.referee_id].used += 1
    strongest = sorted(referees, key=_strength)
    made: list[Assignment] = []
    short_apps: list[ApplicationNeed] = []
    unstated = []
    for need in sorted(needs, key=_order):
        if need.letters is None:
            unstated.append(need.application_id)
            continue
        have = sum(1 for r in load.values() if need.application_id in r.on)
        candidates = sorted(
            (
                r for r in strongest
                if _fits(r, need)
                and need.application_id not in load[r.referee_id].on
                and (need.application_id, r.referee_id) not in blocked
            ),
            key=lambda r: (
                not (need.mode == "UPLOAD_GENERIC" and r.gives_generic_letter),
                not (need.institutional_email and r.institutional_email),
                _strength(r),
            ),
        )
        for referee in candidates:
            if have >= need.letters:
                break
            current = load[referee.referee_id]
            if costs_a_request(need.mode):
                if current.used >= referee.agreed_requests:
                    continue
                current.used += 1
            current.on.add(need.application_id)
            have += 1
            made.append(Assignment(
                need.application_id, referee.referee_id, need.mode,
                email_flag=need.institutional_email and not referee.institutional_email,
            ))
        if have < need.letters:
            short_apps.append(need)
    return Proposal(tuple(made), _short(short_apps, needs, referees), tuple(unstated))


def _short(
    short_apps: Sequence[ApplicationNeed],
    needs: Sequence[ApplicationNeed],
    referees: Sequence[RefereeFacts],
) -> tuple[str, ...]:
    if not short_apps:
        return ()
    if not referees:
        total = sum(n.letters or 0 for n in needs)
        return (f"Your pack needs {total} letters. Add the professors who will write for you.",)
    lines = []
    portal = [n for n in needs if n.letters and costs_a_request(n.mode)]
    needed = sum(n.letters or 0 for n in portal)
    agreed = sum(r.agreed_requests for r in referees)
    if any(costs_a_request(n.mode) for n in short_apps) and needed > agreed:
        willing = [r for r in referees if r.agreed_requests] or list(referees)
        who = (
            f"{willing[0].name} agreed to {agreed}" if len(willing) == 1
            else f"Your referees agreed to {agreed}"
        )
        ordinal = _ORDINALS[len(referees)] if len(referees) < len(_ORDINALS) else "another"
        line = (
            f"{who}; your pack needs {needed} portal letter{'s' if needed != 1 else ''}. "
            f"Ask a {ordinal} referee"
        )
        uploads = sum(1 for n in needs if n.mode == "UPLOAD_GENERIC")
        generic = [r for r in referees if r.gives_generic_letter]
        if uploads and generic:
            whose = f"{generic[0].name}'s" if len(generic) == 1 else "a"
            line += (
                f", or the {uploads} programme{'s' if uploads > 1 else ''} that accept"
                f"{'' if uploads > 1 else 's'} an uploaded letter can use {whose} generic one"
            )
        lines.append(line + ".")
    for need in short_apps:
        if costs_a_request(need.mode) and needed > agreed:
            continue
        kind = {"ACADEMIC": "academic ", "PROFESSIONAL": "professional "}.get(need.kind, "")
        lines.append(
            f"{need.title} needs {need.letters} {kind}letter{'s' if need.letters != 1 else ''}"
            f" from different referees. Add another {kind}referee."
        )
    return tuple(lines)
```

```python
        )
        for r in referees
        if r.status == Referee.Status.ACTIVE
    ]


def _plan_inputs(
    student: Membership, *, keep: tuple[str, ...]
) -> tuple[list[tuple[Application, RefereeNeed, ApplicationNeed]], list[RefereeFacts],
           list[Assignment], list[tuple[UUID, UUID]]]:
    rows = application_needs(student, _open_applications(student))
    referees = list(Referee.objects.filter(student=student).order_by("created_at"))
    requests = RefereeRequest.objects.filter(referee__student=student)
    kept = [
        Assignment(q.application_id, q.referee_id, q.mode)
        for q in requests.filter(state__in=keep, referee__status=Referee.Status.ACTIVE)
    ]
    declined = [
        (q.application_id, q.referee_id)
        for q in requests.filter(state=RefereeRequest.State.DECLINED)
    ]
    return rows, _facts(referees), kept, declined


@transaction.atomic
def repropose(student: Membership) -> Proposal:
    """Replace every PROPOSED request with a fresh proposal; asked ones are kept."""
    _lock(student)  # a second tab waits here, then sees this one's rows
    RefereeRequest.objects.filter(
        referee__student=student, state=RefereeRequest.State.PROPOSED
    ).delete()
    rows, facts, kept, declined = _plan_inputs(student, keep=_KEPT)
    proposal = propose([n for _a, _n, n in rows], facts, kept=kept, declined=declined)
    for made in proposal.assignments:
        RefereeRequest.objects.create(
            referee_id=made.referee_id, application_id=made.application_id,
            mode=made.mode or RefereeRequest.Mode.NOT_STATED,
        )
    return proposal


def _stated(need: RefereeNeed, field: str) -> Stated | None:
    value = getattr(need, field)
    return value if isinstance(value, Stated) else None

@login_required
@require_POST
def referee_request_add(request: HttpRequest) -> HttpResponse:
    """Edit the proposal: put one active professor on one application."""
    student = current_membership(request, Membership.Role.STUDENT)
    try:
        referee_id = UUID(request.POST.get("referee_id", ""))
        application_id = UUID(request.POST.get("application_id", ""))
    except ValueError:
        raise Http404 from None
    referee = get_object_or_404(
        Referee, id=referee_id, student=student, status=Referee.Status.ACTIVE
    )
    # Only a card still waiting for a decision, as _open_applications.
    application = get_object_or_404(Application, id=application_id, student=student, outcome="")
    if _limited(request, student):
        return redirect("student-referee-plan")
    need = application_needs(student, [application])[0][1]
    mode = need.mode.value if need.mode is not None else RefereeRequest.Mode.NOT_STATED
    with transaction.atomic():
        _lock(student)
        made, created = RefereeRequest.objects.get_or_create(
            referee=referee, application=application, defaults={"mode": mode}
        )
        if not created and made.state != RefereeRequest.State.DECLINED:
            messages.error(request, f"{referee.label} is already on {application.title}.")
            return redirect("student-referee-plan")
        if not created:
            # They declined or went silent, and the student puts them back (spec §11.2).
            made.state, made.mode = RefereeRequest.State.PROPOSED, mode
            made.save(update_fields=["state", "mode", "updated_at"])
        record_event(actor=student, event="referee.request_added", object_id=made.id)
    return redirect("student-referee-plan")
```

```python
"""The request package for one professor (lanes spec §11.3). The student sends it.

The platform never emails a third party: this module builds a page and one message the
student copies, and nothing here sends anything. It lists portal NAMES and the senders
the professor should expect, never how to apply (rule 11).
"""

from __future__ import annotations

from dataclasses import dataclass
from urllib.parse import urlsplit
from uuid import UUID

from django.contrib import messages
from django.contrib.auth.decorators import login_required
from django.db.models import F
from django.http import HttpRequest, HttpResponse
from django.shortcuts import get_object_or_404, redirect, render
from django.urls import reverse
from django.views.decorators.http import require_POST

from apps.concierge.models import DocumentVersion
from apps.dashboard.dates import both_zones, counting_deadline, funding_mandatory
from apps.portals.models import Application
from apps.portals.route_snapshot import read_snapshot
from apps.tenancy.models import Membership
from apps.tenancy.services import current_membership

from .audit import record_event
from .models import DocumentDraft, Referee, RefereeRequest, VerifiedFact
from .referee_views import _limited
from .verification import generatable_facts

#: What the student may mark; PROPOSED is the plan's, never a mark.
MARKS = ("ASKED", "AGREED", "DECLINED", "SUBMITTED")
#: What the professor does, per mode, in the message. Never a portal step.
_MODE_WORDS = {
    "PORTAL_INVITE": "the portal will email you an invitation to upload your letter",
    "REFEREE_SENDS": "the programme asks you to send the letter yourself",
    "UPLOAD_GENERIC": "a general letter that I upload is enough",
    "UPLOAD_ADDRESSED": "a letter addressed to the programme, which I upload",
}


@dataclass(frozen=True)
class CvLink:
    label: str
    #: "" when nothing serves the file back to the student (an upload: see Notes).
    url: str = ""


@dataclass(frozen=True)
class PackageLine:
    request: RefereeRequest
    title: str
    deadline: str
    portals: tuple[str, ...]
    senders: tuple[str, ...]
    looks_for: tuple[str, ...]


def package_cv(student: Membership) -> CvLink | None:
    """The newest approved, rendered general CV, else the newest usable CV upload."""
    draft = (
        DocumentDraft.objects.filter(
            student=student, kind=DocumentDraft.Kind.CV, status=DocumentDraft.Status.APPROVED
        )
        .exclude(rendered_filename="")
        .order_by(F("programme_id").asc(nulls_first=True), "-version")
        .first()
    )
    if draft is not None:
        return CvLink(
            f"Your CV (version {draft.version})",
            reverse("documents-download", args=[draft.id, draft.rendered_filename]),
        )
    upload = (
        DocumentVersion.objects.filter(
            student=student, kind="CV", state="AVAILABLE", retired_at__isnull=True
        )
        .order_by("-created_at")
        .first()
    )
    if upload is not None:
        when = upload.created_at
        return CvLink(f"The CV you uploaded on {when.day} {when:%b %Y}")
    return None


def _host(url: str) -> str:
    try:
        return (urlsplit(url).hostname or "").casefold()
    except ValueError:
        return ""


def _portals(application: Application) -> tuple[tuple[str, ...], tuple[str, ...]]:
    """(portal names, sender hosts) from the card's own copies, never the catalogue."""
    names: list[str] = []
    hosts: list[str] = []
    if application.portal_form_id is not None:
        assert application.portal_form is not None
        hosts.append(application.portal_form.portal_host)
    snapshot = read_snapshot(application)
    for step in snapshot.steps if snapshot is not None else []:
        if step.portal_system and step.portal_system not in names:
            names.append(step.portal_system)
        host = _host(step.portal_url)
        if host and host not in hosts:
            hosts.append(host)
    return tuple(names or hosts), tuple(hosts)


def _fact_line(fact: VerifiedFact) -> str:
    p = fact.payload if isinstance(fact.payload, dict) else {}
    if fact.category == VerifiedFact.Category.EDUCATION:
        parts = [
            p.get("degree_english") or p.get("degree_as_awarded"),
            p.get("institution_english") or p.get("institution_original"),
            "–".join(str(d) for d in (p.get("start_date"), p.get("end_date")) if d),
            f"grade {p['grade_native']}" if p.get("grade_native") else "",
        ]
    else:
        parts = [
            p.get("thesis_title"),
            f"supervised by {p['supervisor']}" if p.get("supervisor") else "",
            p.get("institution"),
        ]
    return ", ".join(str(part) for part in parts if part)


def package_lines(student: Membership, referee: Referee) -> list[PackageLine]:
    mandatory = funding_mandatory(student)
    lines = []
    for request in (
        # Cards still waiting for a decision only: a withdrawn or decided card is not
        # something to ask a professor for (as referee_views._open_applications).
        referee.requests.exclude(state=RefereeRequest.State.DECLINED)
        .filter(application__outcome="")
        .select_related("application__candidate", "application__portal_form")
        .order_by("application__created_at")
    ):
        application = request.application
        counted = counting_deadline(application, mandatory=mandatory)
        if counted is None:
            deadline = ""
        elif counted.at is None:
            deadline = counted.wording
        else:
            deadline = both_zones(counted.at, counted.zone, student.timezone)
        portals, senders = _portals(application)
        lines.append(PackageLine(
            request=request,
            title=application.title,
            deadline=deadline,
            portals=portals,
            senders=senders,
            looks_for=tuple(
                text for text in application.kit_requirements if isinstance(text, str)
            ),
        ))
    return lines


def package_message(
    student: Membership, referee: Referee, lines: list[PackageLine], cv: CvLink | None
) -> str:
    """One ready-to-send message, in the student's voice. The student sends it."""
    body = [f"Dear {referee.label},", "",
            "Thank you for supporting my applications. Here is where each letter goes:"]
    for line in lines:
        detail = [line.deadline and f"due {line.deadline}",
                  _MODE_WORDS.get(line.request.mode, "")]
        text = "; ".join(part for part in detail if part)
```

```python
"""One document list per student (lanes spec §10): every card's needs merged into one
entry per document, with the applications that need it, when it is first needed, and
whether what is held is still valid then.

`merge` is pure. `document_list` is the one function that reads, and it reads tenant
rows only: the requirement sentences are the cards' own copies (`kit_requirements`),
the referee facts the candidate's copy (`referee_requirements`). Never stored (spec §3.2).
"""

from __future__ import annotations

from collections.abc import Sequence
from dataclasses import dataclass
from datetime import UTC, datetime
from typing import TYPE_CHECKING
from uuid import UUID

from apps.dashboard.dates import counting_deadline, funding_mandatory
from apps.portals.models import Application, zone_named
from apps.portals.route_snapshot import read_snapshot

from .referee_needs import needs_for
from .table import ROWS, SENDABLE, ApplicationNeeds, Held, needed_kinds, vouched

if TYPE_CHECKING:
    from apps.mailbox.models import MailSignal
    from apps.tenancy.models import Membership

_REFERENCE = "REFERENCE_LETTER"
_EXPIRING = frozenset({"PASSPORT", "LANGUAGE_CERTIFICATE"})


@dataclass(frozen=True)
class CardNeeds:
    application_id: UUID
    title: str
    #: Kit requirements, vouched mail requests and offer conditions, in their own words.
    sentences: tuple[str, ...]
    #: How many reference letters the card's referee facts ask for; None when unstated.
    letters: int | None = None
    #: The card's counting deadline (dashboard/dates.counting_deadline) and its zone.
    deadline: datetime | None = None
    zone: str = ""


@dataclass(frozen=True)
class DocumentEntry:
    kind: str
    label: str
    #: The applications that need it, earliest deadline first.
    titles: tuple[str, ...]
    #: The ones that ask for a certified copy.
    certified: tuple[str, ...] = ()
    #: REFERENCE_LETTER only: the most letters any one application asks for.
    letters: int | None = None
    needed_by: str = ""
    held: bool = False
    validity: str = ""

    @property
    def headline(self) -> str:
        count = len(self.titles)
        return f"{self.label} — needed by {count} application{'s' if count != 1 else ''}"


def _day(at: datetime, zone: str) -> datetime:
    return at.astimezone(zone_named(zone) or UTC)


def _text(at: datetime, zone: str) -> str:
    local = _day(at, zone)
    return f"{local.day} {local:%b}"


def _by_deadline(card: CardNeeds) -> tuple[bool, datetime]:
    return (card.deadline is None, card.deadline or datetime.max.replace(tzinfo=UTC))


def merge(cards: Sequence[CardNeeds], held: Sequence[Held]) -> tuple[DocumentEntry, ...]:
    first = min((c.deadline for c in cards if c.deadline is not None), default=None)
    entries = []
    for kind, label in ROWS:
        needing: list[tuple[CardNeeds, bool]] = []
        for card in sorted(cards, key=_by_deadline):
            wanted = needed_kinds(ApplicationNeeds(card.application_id, card.title, card.sentences))
            if kind in wanted or (kind == _REFERENCE and card.letters):
                needing.append((card, wanted.get(kind, False)))
        if not needing:
            continue
        lead = needing[0][0]
        needed_by = ""
        if lead.deadline is not None:
            needed_by = f"needed by {_text(lead.deadline, lead.zone)} for {lead.title}"
            if lead.deadline == first:
                needed_by += " (your first deadline)"
        mine = [h for h in held if h.kind == kind and h.state in SENDABLE]
        entries.append(DocumentEntry(
            kind=kind,
            label=label,
            titles=tuple(card.title for card, _c in needing),
            certified=tuple(card.title for card, certified in needing if certified),
            letters=max((c.letters for c, _ in needing if c.letters), default=None)
            if kind == _REFERENCE else None,
            needed_by=needed_by,
            held=bool(mine),
            validity=_validity(label, mine, [card for card, _c in needing]),
        ))
    return tuple(entries)


def _validity(label: str, held: Sequence[Held], cards: Sequence[CardNeeds]) -> str:
    """The first card a held document's expiry falls before, in the spec's sentence.
    A document with no expiry date typed is never flagged: nothing is guessed."""
    expiring = [h for h in held if h.kind in _EXPIRING and h.expires_on is not None]
    if not expiring:
        return ""
    expires = max(h.expires_on for h in expiring if h.expires_on is not None)
    for card in cards:
        if card.deadline is not None and _day(card.deadline, card.zone).date() > expires:
            return (
                f"Your {label.lower()} expires {expires.day} {expires:%b}; "
                f"{card.title} needs it valid on {_text(card.deadline, card.zone)}"
            )
    return ""


def _offer_conditions(application: Application) -> tuple[str, ...]:
    """A conditional offer's conditions (spec §10's "offer conditions"): recorded on the
    card only from trusted mail (portals/outcome.py), so each one is a need."""
    signal = application.outcome_evidence
    if application.outcome != Application.Outcome.ADMITTED_CONDITIONAL or signal is None:
        return ()
    raw = signal.outcome_conditions
    if not isinstance(raw, list):  # an erased signal's JSON is {}
        return ()
    return tuple(condition for condition in raw if isinstance(condition, str))


def document_list(
    student: Membership,
    applications: Sequence[Application],
    signals: Sequence[MailSignal],
    held: Sequence[Held],
) -> tuple[DocumentEntry, ...]:
    """The list from the student's own rows: the same applications, mail and held
    documents the by-application table is built from (views.page_context)."""
    asked: dict[UUID, list[str]] = {}
    for signal in signals:
        if signal.application_id is None:
            continue
        if signal.label == "DOCUMENT_REQUEST" and signal.document_name and vouched(signal):
            asked.setdefault(signal.application_id, []).append(signal.document_name)
    mandatory = funding_mandatory(student)
    cards = []
    for application in applications:
        candidate = application.candidate
        records = candidate.referee_requirements if candidate is not None else []
        snapshot = read_snapshot(application)
        need = needs_for(
            records if isinstance(records, list) else [],
            route_kind=snapshot.kind if snapshot is not None else "",
            award=candidate is not None and candidate.opportunity_id is not None,
        )
        counted = counting_deadline(application, mandatory=mandatory)
        cards.append(CardNeeds(
            application_id=application.id,
            title=application.title,
            sentences=(
                *(text for text in application.kit_requirements if isinstance(text, str)),
                *asked.get(application.id, []),
                *_offer_conditions(application),
            ),
            letters=need.letters,
            deadline=counted.at if counted is not None else None,
            zone=counted.zone if counted is not None else "",
        ))
    return merge(cards, held)
```

Each has a default the plan uses until you answer.

1. **The "too few" sentence's generic clause.** Spec §11.2 writes "can use his generic one"; no gender is stored. **Default:** name the professor ("can use Prof. Aliyev's generic one"), or "a generic one" when several give one. The alternative is "their generic one".

The alternative frees those requests for other applications.
